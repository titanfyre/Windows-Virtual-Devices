cmake_minimum_required(VERSION 3.24)
project(WindowsVirtualDevicesJsonSDK LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(PRODUCT_NATIVE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../packaging/native" CACHE PATH "Canonical SDK source")
if(NOT EXISTS "${PRODUCT_NATIVE_SOURCE}/CMakeLists.txt")
 set(PRODUCT_NATIVE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../Windows-Virtual-Devices-CPP-vcpkg")
endif()
option(PRODUCT_TEST_BACKEND "Synthetic tests; no physical device" OFF)
add_library(WindowsVirtualDevicesJsonSDK SHARED Bridge.cpp)
target_compile_features(WindowsVirtualDevicesJsonSDK PRIVATE cxx_std_17)
target_compile_options(WindowsVirtualDevicesJsonSDK PRIVATE /utf-8 /EHsc /W4)
target_compile_definitions(WindowsVirtualDevicesJsonSDK PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(WindowsVirtualDevicesJsonSDK PRIVATE "${PRODUCT_NATIVE_SOURCE}/sdk")
if(PRODUCT_TEST_BACKEND)
 target_sources(WindowsVirtualDevicesJsonSDK PRIVATE tests/FakeSdk.cpp)
else()
 add_subdirectory("${PRODUCT_NATIVE_SOURCE}" native)
 target_link_libraries(WindowsVirtualDevicesJsonSDK PRIVATE WindowsVirtualDevicesSDK::sdk)
endif()
